cmake_minimum_required(VERSION 3.16)
project(port-scanner LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This application requires Windows.")
endif()
add_executable(port-scanner WIN32 main.cpp scanner.cpp)
target_compile_features(port-scanner PRIVATE cxx_std_17)
target_link_libraries(port-scanner PRIVATE ws2_32 comdlg32 comctl32)
if(MSVC)
    target_compile_options(port-scanner PRIVATE /W4 /utf-8 /EHsc /permissive-)
else()
    target_compile_options(port-scanner PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(MINGW)
    target_link_options(port-scanner PRIVATE -municode)
endif()
